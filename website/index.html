React Client (<Si18nProvider> & useTranslation) 
"use client";

import Si18nBrowser from "si18n.js/browser";
import { Si18nProvider, useTranslation } from "si18n.js/react";

const i18n = new Si18nBrowser();

function LanguagePicker() {
  const { t, locale, locales, setLocale, isReady, error } = useTranslation();

  if (error) return <p>Erreur : {error.message}</p>;
  if (!isReady) return <p>Chargement des traductions...</p>;

  return (
    <div>
      <h1>{t("welcome")}</h1>
      <select value={locale} onChange={(e) => void setLocale(e.target.value)}>
        {locales.map((lang) => (
          <option key={lang} value={lang}>{lang.toUpperCase()}</option>
        ))}
      </select>
    </div>
  );
}

export default function App() {
  return (
    <Si18nProvider
      i18n={i18n}
      options={{
        path: "./locales",
        availableLocales: ["fr", "en"],
        fallbackLang: "fr"
      }}
    >
      <LanguagePicker />
    </Si18nProvider>
  );
}
